make:
sudo nano /etc/asound.conf

then add:



# --- Software mixer (dmix) feeding the hardware ---
pcm.dmixed {
  type dmix
  ipc_key 1024
  ipc_perm 0666
  slave {
    # Change if your card is named or indexed differently
    pcm "hw:Device,0"
    # Your DAC’s supported rate (use 48000 or 44100)
    rate 48000
    period_time 0
    period_size 1024
    buffer_size 4096
  }
  bindings {
    0 0
    1 1
  }
}

# --- Plug wrapper adds format conversion on top of dmix ---
pcm.plugdmix {
  type plug
  slave.pcm "dmixed"
}
